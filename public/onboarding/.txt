    const matrix = (function () {
      const canvas = document.getElementById('matrix-canvas');
      const ctx = canvas.getContext('2d');
      const chars = '01';
      const fontSize = 16;
      const FRAME_MS = 40;
      let cols, drops, last = 0;
      let color = '#7c3aed';
      let glow = '#7c3aed';

      function readAccent() {
        const v = getComputedStyle(document.documentElement).getPropertyValue('--accent').trim();
        color = v || '#7c3aed';
        glow = color;
      }

      function resize() {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
        cols = Math.floor(canvas.width / fontSize);
        drops = Array.from({ length: cols }, () => Math.random() * -80);
      }
      window.addEventListener('resize', resize);
      resize();
      readAccent();

      function draw(now) {
        requestAnimationFrame(draw);
        if (now - last < FRAME_MS) return;
        last = now;

        ctx.shadowBlur = 0;
        ctx.fillStyle = 'rgba(8, 8, 8, 0.07)';
        ctx.fillRect(0, 0, canvas.width, canvas.height);

        ctx.font = fontSize + 'px JetBrains Mono, monospace';
        ctx.shadowBlur = 10;
        ctx.shadowColor = glow;
        ctx.fillStyle = color;

        for (let i = 0; i < drops.length; i++) {
          const x = i * fontSize;
          const y = drops[i] * fontSize;
          ctx.fillText(chars[Math.floor(Math.random() * chars.length)], x, y);

          if (y > canvas.height && Math.random() > 0.975) {
            drops[i] = 0;
          } else {
            drops[i]++;
          }
        }
        ctx.shadowBlur = 0;
      }
      requestAnimationFrame(draw);

      return { readAccent };
    })();

    
    const TOTAL_STEPS = 3;

    function buildDots(step) {
      const el = document.getElementById('dots-' + step);
      if (!el) return;
      el.innerHTML = '';
      for (let i = 1; i <= TOTAL_STEPS; i++) {
        const d = document.createElement('div');
        d.className = 'dot' + (i === step ? ' active' : '');
        el.appendChild(d);
      }
    }
    for (let s = 1; s <= TOTAL_STEPS; s++) buildDots(s);

    
    function goTo(index) {
      document.querySelectorAll('.card').forEach(c => {
        c.classList.remove('active');
        if (parseInt(c.dataset.index, 10) === index) {
          setTimeout(() => c.classList.add('active'), 50);
        }
      });
    }

    const params = new URLSearchParams(window.location.search);
    if (params.has('index')) {
      const i = parseInt(params.get('index'), 10);
      if (!isNaN(i) && i >= 0 && i <= 4) goTo(i);
    }

    
    function setTheme(theme, btn) {
      const link = document.getElementById('csslink');
      
      link.onload = () => matrix.readAccent();
      link.href = '/assets/css/colors/' + theme + '.css';
      document.querySelectorAll('.theme-buttons .opt-btn').forEach(b => b.classList.remove('active'));
      if (btn) btn.classList.add('active');
      localStorage.setItem('singularity_theme', theme);
    }

    function setCloak(cloak, btn) {
      const c = CLOAKS[cloak];
      if (!c) return;
      localStorage.setItem('singularity_cloakTitle', c.title);
      localStorage.setItem('singularity_cloakIcon', c.icon);
      localStorage.setItem('singularity_cloakIndex', c.index);
      document.title = c.title;
      document.getElementById('siteicon').href = c.icon;
      document.querySelectorAll('.cloak-buttons .opt-btn').forEach(b => b.classList.remove('active'));
      if (btn) btn.classList.add('active');
    }

    function autoCloak() {
      const win = window.open('about:blank', '_blank');
      if (!win) {
        alert('Popup blocked. Allow popups for this site and try again.');
        return;
      }

      localStorage.setItem('singularity_autoCloak', 'yes');
      localStorage.setItem('singularity_completedOnboarding', 'yes');
      localStorage.setItem('singularity_clearCache', 'yes');

      const title = localStorage.getItem('singularity_cloakTitle') || 'Singularity';
      const iconPath = localStorage.getItem('singularity_cloakIcon') || '/assets/img/fav.png';
      const icon = new URL(iconPath, window.location.origin).href; 

      win.document.title = title;
      const link = win.document.createElement('link');
      link.rel = 'icon';
      link.href = icon;
      win.document.head.appendChild(link);

      win.document.body.style.margin = '0';
      const frame = win.document.createElement('iframe');
      frame.src = window.location.origin + '/';
      Object.assign(frame.style, {
        border: '0',
        position: 'fixed',
        left: '0',
        top: '0',
        width: '100vw',
        height: '100vh'
      });
      win.document.body.appendChild(frame);

      window.location.replace('about:blank');
    }

    function continueToSite() {
      localStorage.setItem('singularity_clearCache', 'yes');
      localStorage.setItem('singularity_completedOnboarding', 'yes');
      window.location.href = '/';
    }
